```rust
pub fn update_garden(id: &str, name: &str) {
    let now = chrono::Utc::now().timestamp_millis();
    let mut gardens = GARDENS.write();
    if let Some(garden) = gardens.iter_mut().find(|g| g.base.id == id) {
        garden.name = name.into();
        garden.base.updated_at = now;
    }
    save_to_storage();
}

pub fn delete_garden(id: &str) {
    let now = chrono::Utc::now().timestamp_millis();
    let mut gardens = GARDENS.write();
    if let Some(garden) = gardens.iter_mut().find(|g| g.base.id == id) {
        garden.base.deleted_at = Some(now);
    }
    // Also soft-delete associated beds
    drop(gardens);
    let mut beds = BEDS.write();
    for bed in beds.iter_mut().filter(|b| b.garden_id == id) {
        bed.base.deleted_at = Some(now);
    }
    save_to_storage();
}
```
